Code Sample & Concept Explanation

Data Binding System:
  • Interpolation & Property Binding: Flow values unidirectionally from the TypeScript component to the HTML view template.
  • Event Binding: Flows data from the HTML view template (user action) to the TypeScript component method.
  • Two-Way Binding: Synchronizes both directions simultaneously, typically using the [(ngModel)] directive (requires FormsModule).

1. Interpolation & Property Flow

Injecting strings dynamically and binding properties to DOM targets.

<!-- HTML Template Usage -->
<!-- Interpolation -->
<div class="binding-item">
  <span class="binding-label">Static App Title:</span>
  <span class="binding-value">{{ appName }}</span>
</div>

<div class="binding-item">
  <span class="binding-label">Dynamic Time Tracker:</span>
  <span class="binding-value text-glowing">{{ currentTimestamp | date:'medium' }}</span>
</div>

<div class="binding-item">
  <span class="binding-label">Mathematical Calculation (15 * 6):</span>
  <span class="binding-value">{{ 15 * 6 }}</span>
</div>

<div class="binding-item">
  <span class="binding-label">Logical Evaluation:</span>
  <span class="binding-value">{{ appName.includes('Showcase') ? 'Showcase detected' : 'Standard app' }}</span>
</div>

<!-- Property Binding -->
<mat-slide-toggle [checked]="isButtonDisabled()" (change)="toggleButtonState()">
  Disable Targeted Button
</mat-slide-toggle>

<button mat-flat-button color="primary" [disabled]="isButtonDisabled()">
  Target Button (Property Bound)
</button>

<mat-slider min="40" max="150" step="5">
  <input matSliderThumb [value]="boxSize()" (valueChange)="boxSize.set($event)" />
</mat-slider>

<div class="preview-box"
     [style.width.px]="boxSize()"
     [style.height.px]="boxSize()"
     [style.backgroundColor]="colorPalette()">
  BOX
</div>

<!-- TypeScript Component Controller -->
public appName = 'Angular Showcase Lab';
public currentTimestamp = new Date();

public isButtonDisabled = signal(false);
public boxSize = signal(60);
public colorPalette = signal('#3f51b5');

public toggleButtonState(): void {
  this.isButtonDisabled.update(state => !state);
}

public changeColor(color: string): void {
  this.colorPalette.set(color);
}

2. Event & Two-Way Sync

Routing actions back to state and setting up bi-directional model sync.

<!-- HTML Template Usage -->
<!-- Event Binding -->
<button mat-flat-button color="accent" (click)="handleButtonClick($event)">
  Increment Click counter (Event Bound)
</button>

<div class="binding-item">
  <span class="binding-label">Total Button Click Actions:</span>
  <span class="binding-value value-glow">{{ clickCount() }}</span>
</div>

<div class="coordinate-tracker-box" (mousemove)="handleMouseMove($event)">
  <span class="tracker-label">Hover mouse over this canvas surface</span>
  <div class="coords">
    X: <span class="coord-val">{{ lastMouseCoords().x }}</span> | 
    Y: <span class="coord-val">{{ lastMouseCoords().y }}</span>
  </div>
</div>

<!-- Two-Way Binding -->
<input [(ngModel)]="userName" />
<div class="avatar-large">
  {{ userName().substring(0, 2).toUpperCase() }}
</div>
<div class="avatar-name">{{ userName() }}</div>

<!-- TypeScript Component Controller -->
public clickCount = signal(0);
public lastMouseCoords = signal({ x: 0, y: 0 });
public userName = signal('Developer');
public userProfileTheme = signal('indigo');

public handleButtonClick(event: MouseEvent): void {
  this.clickCount.update(c => c + 1);
}

public handleMouseMove(event: MouseEvent): void {
  this.lastMouseCoords.set({ x: event.offsetX, y: event.offsetY });
}
