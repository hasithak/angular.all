Code Sample & Concept Explanation

Directives and Pipes Extension Points:
  • Attribute Directives: Modify the behavior, appearance, or layout of existing DOM nodes. Use @HostListener to attach event handlers.
  • Structural Directives: Modify the DOM layout by adding or removing entire HTML elements dynamically. They require TemplateRef (what to render) and ViewContainerRef (where to render).
  • Custom Pipes: Pure functions that transform raw values into formatted displays. Set pure: true (default) so that transformations only run when inputs change.

1. Highlight Attribute Directive

Using the attribute directive in templates, followed by its TS HostListener and HostBinding implementation.

<!-- HTML Template Usage -->
<div [appHighlight]="customHighlightColor()" class="highlight-target-area">
  Hover your cursor over this card area!
</div>

<!-- TypeScript Implementation -->
import { Directive, ElementRef, HostBinding, HostListener, Input, Renderer2 } from '@angular/core';

@Directive({
  selector: '[appHighlight]',
  standalone: true
})
export class HighlightDirective {
  @Input() defaultColor = 'transparent';
  @Input('appHighlight') highlightColor = 'rgba(63, 81, 181, 0.15)';

  @HostBinding('style.backgroundColor') backgroundColor!: string;
  @HostBinding('style.transition') transition = 'background-color 0.3s ease, transform 0.2s ease';
  @HostBinding('style.borderRadius') borderRadius = '8px';

  constructor(
    private el: ElementRef,
    private renderer: Renderer2
  ) {}

  @HostListener('mouseenter') 
  public onMouseEnter(): void {
    this.backgroundColor = this.highlightColor;
    this.renderer.setStyle(this.el.nativeElement, 'transform', 'translateY(-2px)');
  }

  @HostListener('mouseleave') 
  public onMouseLeave(): void {
    this.backgroundColor = this.defaultColor;
    this.renderer.setStyle(this.el.nativeElement, 'transform', 'translateY(0)');
  }
}

2. Custom Structural Directive

Using the structural unless directive in templates, followed by its TS ViewContainerRef implementation.

<!-- HTML Template Usage -->
<div *appUnless="hideUnlessCondition()" class="alert-box green">
  This node is rendered because *appUnless is falsy!
</div>

<!-- TypeScript Implementation -->
import { Directive, Input, TemplateRef, ViewContainerRef } from '@angular/core';

@Directive({
  selector: '[appUnless]',
  standalone: true
})
export class UnlessDirective {
  private hasView = false;

  constructor(
    private templateRef: TemplateRef<any>,
    private viewContainer: ViewContainerRef
  ) {}

  @Input() set appUnless(condition: boolean) {
    if (!condition && !this.hasView) {
      this.viewContainer.createEmbeddedView(this.templateRef);
      this.hasView = true;
    } else if (condition && this.hasView) {
      this.viewContainer.clear();
      this.hasView = false;
    }
  }
}

3. Custom Sorting & Formatting Pipes

Applying Sort and Truncate transforms in templates, followed by their PipeTransform implementations.

<!-- HTML Template Usage -->
<mat-card *ngFor="let prod of products() | sort:sortByProperty():sortOrderDirection()">
  <p>{{ prod.description | truncate:45 }}</p>
  <span>{{ prod.price | currency:'USD' }}</span>
  <span>{{ prod.discount | percent }}</span>
</mat-card>

<!-- TypeScript Sort Pipe Implementation -->
@Pipe({ name: 'sort', standalone: true, pure: true })
export class SortPipe implements PipeTransform {
  public transform<T>(array: T[] | null, property: keyof T, direction: 'asc' | 'desc' = 'asc'): T[] {
    if (!array || !Array.isArray(array)) return [];
    const sorted = [...array].sort((a, b) => {
      const valA = a[property];
      const valB = b[property];
      if (valA === valB) return 0;
      if (typeof valA === 'string' && typeof valB === 'string') return valA.localeCompare(valB);
      return (valA as any) > (valB as any) ? 1 : -1;
    });
    return direction === 'desc' ? sorted.reverse() : sorted;
  }
}

<!-- TypeScript Truncate Pipe Implementation -->
@Pipe({ name: 'truncate', standalone: true })
export class TruncatePipe implements PipeTransform {
  public transform(value: string | null | undefined, limit = 20, trail = '...'): string {
    if (!value) return '';
    return value.length <= limit ? value : value.substring(0, limit) + trail;
  }
}
