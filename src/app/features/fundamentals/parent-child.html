Code Sample & Concept Explanation

Architectural Guidelines:
  • Unidirectional Data Flow: Data always flows down via properties, and events bubble up via actions. This makes tracking states simple and predictable.
  • Encapsulation: While @ViewChild is useful for form resetters or animations, try to avoid tight coupling between parent and child logic.

1. Parent HTML Template

<!-- Input field to set the message -->
<input [(ngModel)]="messageText" placeholder="Type a message..." />

<!-- Binding Input state and catching Output events -->
<app-fundamentals-child
  [parentMessage]="messageText()"
  (counterChange)="handleCounterChange($event)"
  (alertTriggered)="handleChildAlert($event)"
></app-fundamentals-child>

2. Parent Component TS

import { ViewChild, signal } from '@angular/core';

public messageText = signal('Greetings from the Parent Component!');
public latestChildCounter = signal(0);

// Query child reference
@ViewChild(FundamentalsChildComponent) 
public primaryChild!: FundamentalsChildComponent;

public handleCounterChange(count: number): void {
  this.latestChildCounter.set(count);
}

public handleChildAlert(msg: string): void {
  this.snackBar.open(`[Parent Received Event] ${msg}`, 'Dismiss');
}

// Invoke method on child component
public resetChildCounters(): void {
  if (this.primaryChild) {
    this.primaryChild.resetChildState();
    this.messageText.set('');
    this.latestChildCounter.set(0);
  }
}

4. Child Component TS

import { Input, Output, EventEmitter, signal } from '@angular/core';

export class ChildComponent {
  @Input() parentMessage = '';
  
  @Output() counterChange = new EventEmitter<number>();
  @Output() alertTriggered = new EventEmitter<string>();

  public counter = signal(0);

  public incrementCounter(): void {
    this.counter.update(c => c + 1);
    this.counterChange.emit(this.counter());
  }

  public resetChildState(): void {
    this.counter.set(0);
  }

  public notifyReset(): void {
    this.resetChildState();
    this.alertTriggered.emit('Child component state was reset!');
  }
}
