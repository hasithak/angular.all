Code Sample & Concept Explanation

Modern HttpClient Architecture:
  • Functional Interceptors: In modern Angular, interceptors are lightweight functions (HttpInterceptorFn) instead of classes implementing interfaces.
  • Registration: Configured inside the provideHttpClient(withInterceptors([ ... ])) call in app.config.ts.

1. Request Header Injection (Header Interceptor)

Clones the outbound request to inject custom headers immutably before dispatching.

export const headerInterceptor: HttpInterceptorFn = (req, next) => {
  const customReq = req.clone({
    setHeaders: {
      'X-App-Client': 'AngularFeatureShowcase'
    }
  });
  return next(customReq);
};

2. Response Latency Audit (Logging Interceptor)

Intercepts incoming responses to compute execution time and logs state variables.

export const loggingInterceptor: HttpInterceptorFn = (req, next) => {
  const logger = inject(LoggingService);
  const startTime = Date.now();

  return next(req).pipe(
    finalize(() => {
      const duration = Date.now() - startTime;
      logger.info(`Request ${req.method} ${req.url} took ${duration}ms`);
    })
  );
};

<!-- TypeScript Component Controller -->
this.http.get('https://non-existent-api-dns-error.dev/data').pipe(
  // Automatically re-request 2 times (total 3 attempts)
  retry(2),
  catchError(err => {
    return throwError(() => new Error('Connection failed after 2 retry attempts.'));
  })
).subscribe({
  error: err => this.errorLog.set(err.message)
});
